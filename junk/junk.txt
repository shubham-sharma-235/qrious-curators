/* ════════════════════════════════════════════════
   SCROLL-DRIVEN VIDEO SHRINK
   
   As the user scrolls through #scroll-section:
   - progress 0→0.5: video shrinks from full screen
     to ~50% size with rounded corners + fades hero text
   - progress 0.5→1: video holds size, content below visible
════════════════════════════════════════════════ */
const section   = document.getElementById('scroll-section')
const videoWrap = document.getElementById('videoWrap')
const heroText  = document.getElementById('videoHeroText')
const scrollHint= document.getElementById('scrollHint')
const spFill    = document.getElementById('spFill')
const overlay   = document.getElementById('videoOverlay')

/* Easing helper */
function easeInOut(t) {
  return t < .5 ? 2*t*t : -1+(4-2*t)*t
}
function lerp(a, b, t) { return a + (b-a)*t }
function clamp(v, lo, hi) { return Math.max(lo, Math.min(hi, v)) }

function onScroll() {
  const sTop    = section.offsetTop
  const sHeight = section.offsetHeight
  const winH    = window.innerHeight
  const scrollY = window.scrollY

  /* 0 = top of section visible, 1 = bottom of section */
  const raw = (scrollY - sTop) / (sHeight - winH)
  const p   = clamp(raw, 0, 1)

  /* Progress: 0 → 1 during first 65% of scroll */
  const shrinkP = clamp(p / 0.65, 0, 1)
  const easedP  = easeInOut(shrinkP)

  /* ── Scale: 1 → 0.50 ── */
  const scale = lerp(1, 0.50, easedP)

  /* ── Border radius: 0 → 24px ── */
  const radius = lerp(0, 24, easedP)

  /* ── Vertical shift (simulate it rising as shrinks) ── */
  /* We push the wrap upward slightly so it looks centered-ish */
  const translateY = lerp(0, -60, easedP)

  /* Apply */
  videoWrap.style.transform   = `scale(${scale}) translateY(${translateY}px)`
  videoWrap.style.borderRadius = `${radius}px`

  /* ── Hero text fades out ── */
  const textOpacity = clamp(1 - easedP * 2.5, 0, 1)
  heroText.style.opacity   = textOpacity
  heroText.style.transform = `translateY(${easedP * -30}px)`

  /* ── Scroll hint ── */
  scrollHint.style.opacity = clamp(1 - p * 8, 0, 1)

  /* ── Overlay darkens slightly as shrinks ── */
  overlay.style.opacity = lerp(1, 0.6, easedP)

  /* ── Progress rail ── */
  spFill.style.height = (shrinkP * 100) + '%'
}

window.addEventListener('scroll', onScroll, { passive: true })
onScroll() // run once on load

/* ════════════════════════════════════════════════
   FADE-IN OBSERVER (content section)
   Each .fi element animates in when it enters viewport
════════════════════════════════════════════════ */
const fiObs = new IntersectionObserver(entries => {
  entries.forEach(e => {
    if (e.isIntersecting) {
      e.target.classList.add('in')
      fiObs.unobserve(e.target)
    }
  })
}, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' })

document.querySelectorAll('.fi, .fi-left, .fi-right, .fi-scale').forEach(el => fiObs.observe(el))

    if (t) { e.preventDefault(); t.scrollIntoView({ behavior: 'smooth' }) }
  })
})
